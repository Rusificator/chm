#include <iostream>
using namespace std;

int main()
{
    setlocale(LC_ALL, "Rus");
    int n;
    cout << "Введите размерность матрицы: ";
    cin >> n;
    double** a = new double* [n];
    for (int i = 0; i < n; i++)
       a[i] = new double[n+1];

    cout << "Введите матрицу коэффициентов A :" << endl;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> a[i][j];
        }
    }

    cout << "Введите вектор правых частей b :" << endl;
    for (int i = 0,j=n; i<n; i++) {
        cin >> a[i][j];
    }



    for (int i = 0;i < n;i++)
    { 
        cout << "\n";
        for (int j = 0;j < n + 1;j++)
            cout << a[i][j] << " ";
    }
   


}
